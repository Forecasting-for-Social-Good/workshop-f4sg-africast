\begin{frame}{What is a time series foundation model?}
\protect\phantomsection\label{what-is-a-time-series-foundation-model}
\begin{itemize}
\tightlist
\item
  Large, pre-trained models (e.g.~Chronos, TimeGPT, Moirai) that
  forecast directly from historical data
\item
  Trained once on huge, diverse collections of time series
\item
  Little or no task-specific training required to forecast a new series
  (``zero-shot'')
\end{itemize}
\end{frame}

\begin{frame}{How do they compare to classical methods?}
\protect\phantomsection\label{how-do-they-compare-to-classical-methods}
\begin{itemize}
\tightlist
\item
  Classical methods (ETS, ARIMA, regression) are fit to \emph{your}
  series
\item
  Foundation models are pre-trained on \emph{many} series, then applied
  to yours
\item
  Trade-offs: convenience and speed vs.~interpretability and control
\end{itemize}
\end{frame}

\begin{frame}{When to use a foundation model}
\protect\phantomsection\label{when-to-use-a-foundation-model}
\begin{itemize}
\tightlist
\item
  Useful for quick baselines, and series with little historical data
\item
  Classical methods often still competitive, and more transparent, once
  tuned
\item
  Consider the operational context (compute, explainability, cost)
\end{itemize}
\end{frame}

\begin{frame}{Summary}
\protect\phantomsection\label{summary}
\begin{itemize}
\tightlist
\item
  Foundation models are a fast-moving area of forecasting research
\item
  A useful addition to the toolkit alongside ETS, ARIMA and regression
\item
  This session gives a high-level introduction; hands-on examples to
  follow
\end{itemize}
\end{frame}
